`timescale 1ns / 1ps

module proc (Data, Resetn, w, Clock, F, Rx, Ry, Done, BusWires);
    input [7:0] Data;
    input Resetn, w, Clock;
    input [1:0] F, Rx, Ry;
    output wire Done;
    output reg [7:0] BusWires;

    localparam [2:0] S_IDLE   = 3'b000,
                     S_FETCH  = 3'b001,
                     S_DECODE = 3'b010,
                     S_EXEC   = 3'b011,
                     S_WRITE  = 3'b100;

    localparam [1:0] OP_LOAD = 2'b00,
                     OP_MOVE = 2'b01,
                     OP_ADD  = 2'b10,
                     OP_SUB  = 2'b11;

    reg [2:0] state;
    reg [1:0] opcode, destination, source;
    reg [7:0] immediate, A, G;
    reg [7:0] registers [0:3];
    integer i;

    assign Done = (state == S_WRITE);

    always @(*) begin
        BusWires = 8'b0;

        case (state)
            S_EXEC: begin
                if (opcode == OP_LOAD)
                    BusWires = immediate;
                else
                    BusWires = registers[source];
            end
            S_WRITE: BusWires = G;
            default: BusWires = 8'b0;
        endcase
    end

    always @(posedge Clock or negedge Resetn) begin
        if (!Resetn) begin
            state       <= S_IDLE;
            opcode      <= OP_LOAD;
            destination <= 2'b00;
            source      <= 2'b00;
            immediate   <= 8'b0;
            A           <= 8'b0;
            G           <= 8'b0;

            for (i = 0; i < 4; i = i + 1)
                registers[i] <= 8'b0;
        end else begin
            case (state)
                S_IDLE: begin
                    if (w)
                        state <= S_FETCH;
                end

                S_FETCH: begin
                    opcode      <= F;
                    destination <= Rx;
                    source      <= Ry;
                    immediate   <= Data;
                    state       <= S_DECODE;
                end

                S_DECODE: begin
                    if ((opcode == OP_ADD) || (opcode == OP_SUB))
                        A <= registers[destination];
                    state <= S_EXEC;
                end

                S_EXEC: begin
                    case (opcode)
                        OP_LOAD: G <= BusWires;
                        OP_MOVE: G <= BusWires;
                        OP_ADD:  G <= A + BusWires;
                        OP_SUB:  G <= A - BusWires;
                        default: G <= 8'b0;
                    endcase
                    state <= S_WRITE;
                end

                S_WRITE: begin
                    registers[destination] <= BusWires;
                    state <= S_IDLE;
                end

                default: state <= S_IDLE;
            endcase
        end
    end
endmodule

module tb_proc;
    reg [7:0] Data;
    reg Resetn;
    reg w;
    reg Clock;
    reg [1:0] F;
    reg [1:0] Rx;
    reg [1:0] Ry;

    wire [7:0] BusWires;
    wire Done;

    proc uut (
        .Data(Data),
        .Resetn(Resetn),
        .w(w),
        .Clock(Clock),
        .F(F),
        .Rx(Rx),
        .Ry(Ry),
        .Done(Done),
        .BusWires(BusWires)
    );

    always #5 Clock = ~Clock;

    task run_instruction;
        input [1:0] instruction;
        input [1:0] destination_register;
        input [1:0] source_register;
        input [7:0] input_data;
        input [7:0] expected_result;
        begin
            @(negedge Clock);
            F = instruction;
            Rx = destination_register;
            Ry = source_register;
            Data = input_data;
            w = 1'b1;

            @(negedge Clock);
            w = 1'b0;

            wait (Done === 1'b1);
            #1;
            if (BusWires !== expected_result)
                $fatal(1, "Expected %0d, got %0d", expected_result, BusWires);

            $display("PASS: opcode %b wrote %0d to R%0d",
                     instruction, BusWires, destination_register);
            @(posedge Clock);
            #1;
        end
    endtask

    initial begin
        Clock = 1'b0;
        Resetn = 1'b1;
        w = 1'b0;
        Data = 8'b0;
        F = 2'b0;
        Rx = 2'b0;
        Ry = 2'b0;

        #2 Resetn = 1'b0;
        #10 Resetn = 1'b1;

        run_instruction(2'b00, 2'd0, 2'd0, 8'd12, 8'd12);
        run_instruction(2'b00, 2'd1, 2'd0, 8'd5,  8'd5);
        run_instruction(2'b10, 2'd0, 2'd1, 8'd0,  8'd17);
        run_instruction(2'b01, 2'd2, 2'd0, 8'd0,  8'd17);
        run_instruction(2'b11, 2'd0, 2'd1, 8'd0,  8'd12);

        $display("All processor tests passed.");
        $finish;
    end

    initial begin
        #2000;
        $fatal(1, "Testbench timed out.");
    end
endmodule
